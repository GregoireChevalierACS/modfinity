# Country scope.

# A scholar only visits countries with some income, and not twice within the cooldown.
modfinity_scholar_can_visit = {
	monthly_income_trade_and_tax > 1
	NOT = { has_variable = modfinity_scholar_cooldown }
}

modfinity_scholar_can_afford_funding = {
	gold >= modfinity_scholar_funding_cost
}

modfinity_scholar_can_afford_praise = {
	gold >= modfinity_scholar_praise_cost
}
